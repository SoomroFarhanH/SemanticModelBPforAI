import { RayfinClient } from '@microsoft/rayfin-client';

import type { UniversalAppSchema } from '../../rayfin/data/schema';
import type { AppFunctionsSchema } from '../../rayfin/functions/src/types';

let _client: RayfinClient<UniversalAppSchema, AppFunctionsSchema> | undefined;

/**
 * Returns the pre-configured RayfinClient singleton.
 */
export function getRayfinClient(): RayfinClient<
  UniversalAppSchema,
  AppFunctionsSchema
> {
  if (!_client) {
    const apiUrl = import.meta.env.VITE_RAYFIN_API_URL;
    const publishableKey = import.meta.env.VITE_RAYFIN_PUBLISHABLE_KEY;

    if (!apiUrl || !publishableKey) {
      throw new Error(
        `Missing required env vars for creating rayfin client - run 'npx rayfin up'`
      );
    }

    _client = new RayfinClient<UniversalAppSchema, AppFunctionsSchema>({
      baseUrl: apiUrl,
      publishableKey,
      authStorage: true,
      useProxy: false,
      functionsBaseUrl: import.meta.env.VITE_RAYFIN_FUNCTIONS_URL,
    });
  }

  return _client;
}
